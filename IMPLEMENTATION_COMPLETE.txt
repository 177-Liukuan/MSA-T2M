━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 检索引导机制
   ├─ 从库中检索Top-3最相似的h_cls（全局表示）
   ├─ 使用原始相似度作为加权系数
   └─ 融合后得到单一检索token，融入Transformer序列

2. 无分类器引导 (CFG) 协同
   ├─ 文本 mask: 10% 概率置零（仅使用检索）
   ├─ 检索 mask: 10% 概率置零（仅使用文本）
   └─ 推理时：x = G(∅,∅) + s·[G(text,ret) - G(∅,∅)]

3. Two-Forward自回归优化
   ├─ 第一阶段: 无梯度预测（"看不见答案"）
   ├─ 第二阶段: 用预测逐步替换原始输入（自回归）
   └─ cosine衰减: 早期保守→晚期激进，缓解曝光偏差

4. 高效架构设计
   ├─ 两个独立投影层（文本512→768, h_cls512→768）
   ├─ 加权融合无需额外学习参数
   └─ 因果掩码保证自回归约束
